CREATE TABLE utenti (
    id INT AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(50) UNIQUE NOT NULL,
    password VARCHAR(255) NOT NULL,
    ruolo ENUM('dottore', 'paziente', 'admin') NOT NULL,
    immagine VARCHAR(255)
);

CREATE TABLE dottori (
    codice_dottore VARCHAR(10) PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
	specializzazione VARCHAR(100),
    id_utente INT UNIQUE NOT NULL,
	FOREIGN KEY (id_utente) REFERENCES utenti(id)
);

CREATE TABLE pazienti (
    codice_fiscale VARCHAR(16) PRIMARY KEY,
	nome VARCHAR(100) NOT NULL,
    data_nascita DATE NOT NULL,
    id_utente INT UNIQUE NOT NULL,
	FOREIGN KEY (id_utente) REFERENCES utenti(id)
);

CREATE TABLE appuntamenti (
	codDoc VARCHAR(10) NOT NULL,
    cfPaz VARCHAR(16) NOT NULL,
	data_appuntamento DATE NOT NULL,
    ora TIME NOT NULL,
    PRIMARY KEY (codDoc, cfPaz),
	FOREIGN KEY (codDoc) REFERENCES dottori(codice_dottore),
	FOREIGN KEY (cfPaz) REFERENCES pazienti(codice_fiscale)
);
